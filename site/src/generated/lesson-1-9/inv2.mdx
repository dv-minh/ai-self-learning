$$
\mathbf{A} = \begin{bmatrix} 1 & 2 \\ 6 & 1 \end{bmatrix}
$$

$$
\det(\mathbf{A}) = 1\cdot 1 - 2\cdot 6 = -11
$$

$$
\mathbf{A}^{-1} = \frac{1}{(-11)}\begin{bmatrix} 1 & -2 \\ -6 & 1 \end{bmatrix} = \begin{bmatrix} -\frac{1}{11} & \frac{2}{11} \\ \frac{6}{11} & -\frac{1}{11} \end{bmatrix}
$$

$$
\mathbf{A}\mathbf{A}^{-1} = \begin{bmatrix} 1 & 2 \\ 6 & 1 \end{bmatrix}\begin{bmatrix} -\frac{1}{11} & \frac{2}{11} \\ \frac{6}{11} & -\frac{1}{11} \end{bmatrix} = \begin{bmatrix} 1 & 0 \\ 0 & 1 \end{bmatrix}
$$
