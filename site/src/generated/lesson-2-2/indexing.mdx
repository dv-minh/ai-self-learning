```python
import numpy as np

A = np.array([[10, 11, 12, 13],
              [20, 21, 22, 23],
              [30, 31, 32, 33]])

print(A[1, 2])      # hàng 1, cột 2 (đếm từ 0)
print(A[-1, -1])    # phần tử cuối
print(A[1][2])      # cho cùng kết quả
print(A[A > 31])    # lọc bằng mặt nạ bool
```
