```python
import numpy as np

A = np.arange(12).reshape(3, 4)
print(A)
print(A[1:, :2])    # từ hàng 1 trở đi, 2 cột đầu
print(A[:, 1])      # cả cột 1 (thành vector 1 chiều)
print(A[::2, ::-1]) # cách 1 hàng, đảo cột
```
