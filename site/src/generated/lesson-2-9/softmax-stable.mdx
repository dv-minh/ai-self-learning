```python
import numpy as np

z = np.array([[1000.0, 1001.0, 1002.0]])

bad = np.exp(z) / np.exp(z).sum(axis=1, keepdims=True)   # e^1000 tràn số
e = np.exp(z - z.max(axis=1, keepdims=True))             # trừ max trước
good = e / e.sum(axis=1, keepdims=True)

print(bad)
print(np.round(good, 6))
```
