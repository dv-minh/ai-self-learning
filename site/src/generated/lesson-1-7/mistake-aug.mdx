$$
\left[\begin{array}{cc|c} -2 & 1 & 3 \\ 4 & 2 & -2 \end{array}\right]
$$

$$
\left[\begin{array}{cc|c} -2 & 1 & 3 \\ 4 & 2 & 4 \end{array}\right]
$$
