```python
import numpy as np

A = np.array([[1, 2, 3],
              [4, 5, 6]])

print(A.sum(axis=0))   # rút gọn theo trục 0: một giá trị cho mỗi cột
print(A.sum(axis=1))   # rút gọn theo trục 1: một giá trị cho mỗi hàng
print(A.sum())         # không chỉ trục: tổng tất cả
```
