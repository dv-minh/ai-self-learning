```python
import numpy as np

a = np.array([[1, 2], [3, 4]])
b = np.array([[5, 6], [7, 8]])

print(np.concatenate([a, b], axis=0).shape)   # nối thêm hàng
print(np.concatenate([a, b], axis=1).shape)   # nối thêm cột
print(np.stack([a, b], axis=0).shape)         # tạo trục MỚI
```
